SYMBOL = "BTCUSDT"
BINANCE_WS_URL = "wss://stream.binance.com:9443/stream"

                order_age_scorer: OrderAgeDistributionScorer,

        self.order_age_scorer = order_age_scorer

        self.alpha_signal_pipeline = alpha_signal_pipeline
        self.alpha_blender = alpha_blender

        self.ws = None

    async def connect(self, retries=100, delay=3):
        streams = f"{self.symbol}@depth@1000ms/{self.symbol}@trade"
        uri = f"{BINANCE_WS_URL}?streams={streams}"
        for attempt in range(retries):
            try:
                self.ws = await websockets.connect(uri, ping_interval=15, ping_timeout=10)
                print(Fore.GREEN + f"[CONNECTED] Binance WebSocket for {self.symbol}")
                return
            except Exception as e:
                print(Fore.RED + f"[CONNECT ERROR] Attempt {attempt+1}/{retries} failed: {e}")
                time.sleep(delay)
        raise ConnectionError(Fore.RED + "Failed to connect to Binance WebSocket after multiple attempts.")

    async def close(self):
        if self.ws:
            await self.ws.close()
            print(Fore.RED + "[DISCONNECTED] WebSocket closed")

# === Entry Point ===
if __name__ == "__main__":
    async def main():
        
        tunerforlayering = CancelWindowTunerForLayering()
        order_layering_tracker = OrderLayeringDetection(tunerforlayering)

        tuner = CancelWindowTuner()

        order_layering_score = LayeringScoring(layering_detector=order_layering_tracker)
        
        order_age_tracker = OrderAgeDistribution()
        order_age_tracker_score = OrderAgeDistributionScorer(tracker=order_age_tracker)

        order_book = OrderBook()
        signal_confidence_calibrator = SignalConfidenceCalibrator()

        cancel_window = SimpleCancelWindow(
            tuner=tuner,
            order_layering=order_layering_tracker,
            order_age_tracker=order_age_tracker,
            order_book=order_book,
            classifier=None #This is done because the modules are co-dependent to prevent the recursive call
        )

        cancel_activity_scorer = CancelActivityScorer(window_ms_tuner=cancel_window)

        classifier = CognitiveMarketRegimeClassifier(
            orderbook=order_book,
            signal_calibrator=signal_confidence_calibrator,
            cancel_window=cancel_window
        )

        # this is done because the modules are co-dependent
        cancel_window.classifier = classifier

        # Alpha Blender
        alpha_blender = AlphaBlender(
            weights = {'cancel_activity': 0.5, 'layering': 0.3, 'order_age': 0.2},
            blending_method = 'weighted average',
            adaptive = True

        )

        #Alpha signal pipeline
        alpha_signal_pipeline = AlphaSignalPipeline(
            cancel_scorer= cancel_activity_scorer,
            age_scorer= order_age_tracker_score,
            layering_scorer= order_layering_score,
            blender = alpha_blender
        )

        runner = BinanceCancelWindowRunner(
            alpha_blender = alpha_blender,
            alpha_signal_pipeline = alpha_signal_pipeline,
            cancel_activity_scorer=cancel_activity_scorer,
            order_layering_scorer=order_layering_score,
            order_age_scorer=order_age_tracker_score,
            cancel_window=cancel_window,
            orderbook=order_book
        )

        await runner.connect()
        try:
            await runner.run()
        finally:
            await runner.close()

    try:
        asyncio.run(main())
    except KeyboardInterrupt:
        print(Fore.RED + "\n[INTERRUPTED] Execution halted.")
